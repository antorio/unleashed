# Python 3.13 compatible (Colab "Latest" runtime, and macOS/CPU).
# torch / torchvision are intentionally NOT pinned: Colab ships a CUDA build
# that already matches its Python, so pinning forced a ~900MB download of a
# wheel that may not even exist for the current interpreter. They are only used
# by the optional Clip2Seg mask engine and a cuda.is_available() check.
numpy==2.1.3
gradio==5.9.1
opencv-python-headless==4.10.0.84
onnx==1.18.0
insightface==0.7.3
psutil==5.9.6
# PyTorch has no builds for Intel Macs beyond 2.2 (NumPy 1 only), so it is
# skipped there; the app runs without it (only the Clip2Seg mask needs it).
torch; platform_system != 'Darwin' or platform_machine != 'x86_64'
torchvision; platform_system != 'Darwin' or platform_machine != 'x86_64'
# 1.27+ is built for CUDA 13, like Colab's torch since Oct 2026 (+cu130);
# 1.21 needed CUDA 12 and ran every model on the CPU there (libcublasLt.so.12
# missing). The CUDA 13 libraries come with torch (nvidia/cu13, nvidia/cudnn).
onnxruntime-gpu==1.30.0; sys_platform != 'darwin'
# macOS has no CUDA build -> plain onnxruntime (CPU / CoreML). Deliberately
# UNPINNED: the newest available version differs by architecture (Apple
# Silicon reaches 1.29, Intel Macs stop at 1.23.2), so any fixed pin breaks
# on one of them. pip picks the highest that exists for the machine.
onnxruntime; sys_platform == 'darwin'
